### JS Additions (~80 lines)

```js
/* Modified celebrate() — perfect score → rainbow palette */
function celebrate() {
  if (typeof window.confetti !== 'function') return;
  var isPerfect = state && state.hearts === 3;
  var palette = isPerfect
    ? ['#EF4444','#F59E0B','#10B981','#06B6D4','#8B5CF6','#EC4899']  // 6-color rainbow
    : ['#06B6D4','#10B981','#F59E0B','#F97316'];                       // default 4-color
  var count = isPerfect ? 120 : 80;
  // fire from both sides
  confetti({ particleCount: count, angle: 60,  spread: 55, origin: {x:0, y:0.7}, colors: palette });
  confetti({ particleCount: count, angle: 120, spread: 55, origin: {x:1, y:0.7}, colors: palette });
}

/* Sparkle rain */
function triggerSparkleRain(count) {
  count = count || 24;
  var reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  for (var i = 0; i < count; i++) {
    var p = document.createElement('div');
    p.className = 'sparkle-particle';
    p.style.left = (Math.random() * 100) + 'vw';
    var size = 4 + Math.random() * 4;
    p.style.width = size + 'px'; p.style.height = size + 'px';
    p.style.animationDelay = (Math.random() * 0.5) + 's';
    p.style.background = ['#F59E0B','#FBBF24','#F97316','#FCD34D'][Math.floor(Math.random()*4)];
    document.body.appendChild(p);
    setTimeout(function(el){return function(){if(el.parentNode)el.parentNode.removeChild(el);};}(p),
               reduced ? 200 : 2000);
  }
}

/* Module / Course completion celebration */
function triggerModuleCelebration() {
  var lessonNum = (state.currentLessonIdx || 0) + 1;
  var isGraduation = (lessonNum === 18);
  var isTrophy = (lessonNum === 6 || lessonNum === 13);
  if (!isGraduation && !isTrophy) return;
  var el = document.createElement('div');
  el.className = isGraduation ? 'graduation-celebration' : 'trophy-celebration';
  el.textContent = isGraduation ? '🎓' : '🏆';
  document.body.appendChild(el);
  var reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  setTimeout(function(){if(el.parentNode)el.parentNode.removeChild(el);}, reduced ? 1000 : 2200);
  if (isGraduation) celebrate();  // B18 always rainbow
}

/* Combined Step 4 success hook */
function triggerStep4Success() {
  celebrate();                       // confetti (rainbow if perfect)
  triggerSparkleRain(24);            // golden sparkle rain
  triggerModuleCelebration();        // trophy B6/B13 OR graduation B18
}
```

#### 2. Diagram MCQ (`format: 'diagram'`)

Per user choice "both" — supports **2 sub-formats**:

- **String (ASCII art)** → `<pre class="mcq-diagram-option">┌──┬──┐...</pre>` — preserves whitespace
- **Array of arrays (HTML table)** → `<table class="mcq-diagram-table"><thead>...</thead><tbody>...</tbody></table>` with cyan headers + gray borders

JS renderer (auto-detect type via `typeof` and `Array.isArray`):
```js
function renderDiagramOption(diagram) {
  if (!diagram) return '<span>(empty)</span>';
  if (typeof diagram === 'string') {
    return `<pre class="mcq-diagram-option">${escapeHtml(diagram)}</pre>`;
  }
  if (Array.isArray(diagram) && diagram.length > 0) {
    // Header row + body rows → HTML table
    let html = '<table class="mcq-diagram-table"><thead><tr>';
    headerRow.forEach(c => html += '<th>' + escapeHtml(String(c)) + '</th>');
    // ... tbody + close
    return `<div class="mcq-diagram-option">${html}</div>`;
  }
  return `<span>${escapeHtml(String(diagram))}</span>`;
}
```

| Decision | Choice | Rationale |
|----------|--------|-----------|
| B18 celebration | **🎓 Graduation cap** (different from trophy) | User explicitly chose "b18-different" — distinct visual graduation theme |

| Sparkle color variants | 4 colors random: gold/amber/orange/yellow | More visual variety than single gold |
| Sparkle size | Random 4-8px | Avoid uniform look — natural rain effect |
| Code MCQ syntax highlighter | Reuse existing `highlightSQL()` | Avoid duplicating tokenizer logic |
| `.t-*` classes scoped inside `.ide-content` originally | Add `.mcq-option .t-*` selectors outside | Lightweight reuse without global pollution |
| Diagram MCQ format | Support both string + array | Per user choice "both" — flexibility for content authors |
