## Part 2 — C7: Progressive Disclosure (ContentCore-inspired)

### Concept

Step 1 (Theory) currently shows ALL content at once (primer + concept cards + diagram + visual DB). With C7:
- **Initial load:** Primer visible (always)
- **Scroll down:** 3 hidden sections (concept cards, diagram, visual-db-panel) fade in via IntersectionObserver
- **Stagger:** Each section delayed by `var(--i) * 150ms`

### CSS (~300 bytes)

```css
.step1-reveal {
  opacity: 0;
  transform: translateY(20px);
  transition:
    opacity var(--dur-base, 200ms) var(--ease-out-quart),
    transform var(--dur-base, 200ms) var(--ease-out-quart);
}

.step1-reveal.is-visible {
  opacity: 1;
  transform: translateY(0);
}

.step1-reveal[data-stagger]:not(.is-visible) {
  transition-delay: calc(var(--i, 0) * 150ms);
}

@media (prefers-reduced-motion: reduce) {
  .step1-reveal {
    opacity: 1 !important;
    transform: none !important;
    transition: none !important;
  }
}
```

### JS Additions

```js
/* Wrap 3 sections + assign --i stagger index */
function wrapStep1RevealSections() {
  var sections = [
    { el: document.getElementById('concept-cards-mount'), i: 0 },
    { el: document.getElementById('primer-svg-mount'),    i: 1 },
    { el: document.getElementById('visual-db-panel'),     i: 2 }
  ];
  sections.forEach(function (s) {
    if (!s.el) return;
    s.el.classList.remove('step1-reveal', 'is-visible');
    s.el.style.removeProperty('--i');
    if (s.el.children.length === 0 && !s.el.textContent.trim()) return;  // skip empty
    s.el.classList.add('step1-reveal');
    s.el.style.setProperty('--i', String(s.i));
  });
  initStep1Reveal();
}

/* IntersectionObserver — trigger fade-in once when 15% visible */
function initStep1Reveal() {
  var sections = document.querySelectorAll('.step1-reveal:not(.is-visible)');
  if (!sections.length) return;
  var reduced = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  if (reduced || typeof IntersectionObserver === 'undefined') {
    sections.forEach(function (el) { el.classList.add('is-visible'); });
    return;
  }
  var observer = new IntersectionObserver(function (entries) {
    entries.forEach(function (e) {
      if (e.isIntersecting) {
        e.target.classList.add('is-visible');
        observer.unobserve(e.target);  // trigger once only
      }
    });
  }, { threshold: 0.15, rootMargin: '0px 0px -50px 0px' });
  sections.forEach(function (s) { observer.observe(s); });
}

// Called at end of renderStep1()
wrapStep1RevealSections();
```

### Behavior

| State | Section class | opacity | transform |
|-------|---------------|---------|-----------|
| Initial (before scroll) | `.step1-reveal` (no `is-visible`) | 0 | translateY(20px) |
| Scrolled into view (≥15%) | `.step1-reveal.is-visible` | 1 | translateY(0) |
| Reduced motion | Always `.is-visible` immediately | 1 | none |
