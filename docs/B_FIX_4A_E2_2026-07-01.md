#### 3.2.1 Token whitelist scan TRƯỚC parser (E3 honest-error)

```js
function scanUnsupportedTokens(sqlText) {
  var s = String(sqlText || '').toLowerCase();
  var checks = [
    { kw: 'IN (',      test: /\bwhere\s+[\w.\s,]+\s+in\s*\(/i,                  msg: 'IN-subquery' },
    { kw: 'CASE ',     test: /\bselect\s+.*\bcase\b/i,                          msg: 'CASE WHEN'  },
    { kw: '->>',       test: /->>/,                                              msg: 'JSONB operator (->>)' },
    { kw: '%s',        test: /=\s*%s|,\s*%s/i,                                  msg: 'psycopg2 placeholder (%s)' },
    { kw: 'ST_DWithin',test: /\bst_dwithin\b/i,                                 msg: 'PostGIS spatial (ST_DWithin)' },
    { kw: 'ST_MakePoint',test: /\bst_makepoint\b/i,                             msg: 'PostGIS spatial (ST_MakePoint)' },
    { kw: '.objects.', test: /\.\s*objects\s*\./i,                              msg: 'Django ORM (.objects.)' },
    { kw: '.values(',  test: /\.\s*values\s*\(/i,                               msg: 'Django ORM (.values())' },
    { kw: '.filter(',  test: /\.\s*filter\s*\(/i,                               msg: 'Django ORM (.filter())' },
  ];
  for (var i = 0; i < checks.length; i++) {
    if (checks[i].test.test(sqlText)) return { unsupported: true, kw: checks[i].msg };
  }
  return { unsupported: false };
}
```

#### 3.2.2 Structured parser — boundary keyword scan

`PE_parseSQLToBlocks` đổi từ regex naive ("FROM <word>" rồi drop everything-after) thành boundary-tracking:

```js
// 4A-E2 fix: fromIdx tìm FIRST "FROM" ngoài parens (Bài 2 EXTRACT(YEAR FROM ...)).
// Track depth — nếu '(' tăng depth, ')' giảm. Tại depth=0 mới match FROM keyword.
var _depth = 0;
for (var _fi = 0; _fi < _scan.length; _fi++) {
  var _ch = _scan.charAt(_fi);
  if (_ch === '(') _depth++;
  else if (_ch === ')') _depth--;
  else if (_depth === 0 && /\sfrom\s/i.test(_scan.substring(_fi))) { fromIdx = _fi; break; }
}

// Tách colsStr (SELECT projection) | afterFrom (FROM body + clauses)

var fromKwRe = /\s+(?:where|group\s+by|having|order\s+by|limit)\b/i;
var fromEnd = fromKwRe.exec(afterFrom);
var fromAndJoins = fromEnd ? afterFrom.substring(0, fromEnd.index).trim() : afterFrom.trim();
var tail = fromEnd ? afterFrom.substring(fromEnd.index).trim() : '';

// Consume WHERE → GROUP BY → HAVING → ORDER BY → LIMIT theo thứ tự
function consumeClause(restStr, kwRegexSrc) {
  var prefix = new RegExp('^' + kwRegexSrc + '\\s+', 'i');
  var m = prefix.exec(restStr);
  if (!m) return { val: null, rest: restStr };
  var body = restStr.substring(m[0].length);
  // Find NEXT clause keyword ANYWHERE in body (NOT ^anchored — body starts after consumed kw).
  var nextKwRe = /(?:where|group\s+by|having|order\s+by|limit)\s+/i;
  var nm = nextKwRe.exec(body);
  if (nm) return { val: body.substring(0, nm.index).trim(), rest: body.substring(nm.index).trim() };
  return { val: body.trim(), rest: '' };
}
```

2 trụ cột:
1. **depth-tracking** để không match `FROM` bên trong parens (Bài 2 EXTRACT regression).
2. **nextKwRe NOT ^anchored** — vì body đã có kw leading stripped, keyword kế tiếp search anywhere. Anchored `^` đã có lỗi nuốt cả GROUP BY vào WHERE.

#### 3.2.9 Aggregation helpers

```js
function detectAggregate(token) {
  var trimmed = String(token || '').trim();
  var m = /^\s*(count|sum|avg|min|max)\s*\((.*)\)\s*(?:\s+as\s+(\w+))?\s*$/i.exec(trimmed);
  if (!m) return null;
  return { fn: m[1].toLowerCase(), expr: m[2].trim(), alias: m[3] || null };
}

function computeAggregate(rows, proj) {
  var fn = proj.fn, expr = proj.expr;
  if (fn === 'count') {
    if (expr === '*') return rows.length;
    // COUNT(expr): đếm non-null/non-empty expr values per row.
    var cnt = 0;
    for (var ri = 0; ri < rows.length; ri++) {
      var v = getRowVal(rows[ri], expr);
      if (v !== undefined && v !== null && v !== '') cnt++;
    }
    return cnt;
  }
  var nums = [];
  for (var rj = 0; rj < rows.length; rj++) {
    var x = evalSqlExpr(rows[rj], expr);
    if (x !== null && !isNaN(x)) nums.push(x);
  }
  if (fn === 'sum') return nums.reduce((a, b) => a + b, 0);
  if (fn === 'avg') return nums.length ? nums.reduce((a, b) => a + b, 0) / nums.length : 0;
  if (fn === 'min') return nums.length ? Math.min.apply(null, nums) : null;
  if (fn === 'max') return nums.length ? Math.max.apply(null, nums) : null;
}
```

### 5.1 Bug: regex `/^\s+from\s+\w+(?:\s+\w+)?/i` eat WHERE keyword as alias

Symptom: Bài 1 SQL `SELECT name, price FROM game_catalog WHERE id = 101;` → parsedTables = ["id"] (misread "id" as table name). 18/20 tests failing "Schema không đầy đủ (cần bảng: id)".

Root cause: regex `(?:\s+(\w+))?` captures WHERE/GROUP/HAVING/ORDER/LIMIT keyword AS alias. Side effect: `replace(/^\s+from\s+\w+(?:\s+\w+)?/i, '')` strips "FROM game_catalog WHERE" → `fromAndJoins = "id = 101"`. THEN `primaryMatch = /^(\w+)/.exec("id = 101") = "id"` → mistreats 'id' as table.

Fix:
1. Strip only `^\s+from\s+` (KHÔNG include alias) — let fromAndJoins flow to fromKwRe naturally.
2. Build aliasMap from `fromAndJoins` scanning (with aliasKwSet filter — bỏ qua WHERE/GROUP/HAVING/ORDER/LIMIT/ON/JOIN/AS/AND keywords as alias candidate).

### 5.2 Bug: `name: t.name` of STRING is undefined

In cross-product loop: `var t = tables[pi]; ... ext[c.name]...` nhưng `t.name` is undefined (table object uses `table_name` key). Replaced with local `name: tableName` (param).

### 5.3 Bug: ON filter alias-prefix mismatch

JoinedRow keys = `orders.col`, `members.col` (full table names). ON condition uses `l.member_id = m.member_id` (aliased). `r['l.member_id']` undefined → join returns 0 rows.

Fix: `resolveOnRef(ref)` — translate alias-prefix to full table name using aliasMap.

### 5.4 Bug: PE_runSQL schema shape mismatch (nested vs top-level)

`runCodeIDE` passes `s4 = state.currentLesson.step_4` with shape `{prompt, starter, schema: {...}, related_schemas: [...]}`. PE_runSQL was looking `schema.table_name` which is undefined at top level → "Schema không đầy đủ".

Fix: `primarySchema = (schema && schema.schema) || schema || {}`; resolve table_name from BOTH sources.

### 5.5 Bug: nextKwRe `^` anchor swallows remaining clauses

Body passed to WHERE consume: `"dlc_no = 2 AND ref_game_id = 300"`. With `^` anchor, nextKwRe.match found nothing → WHERE consumed all of `dlc_no=2 AND ref_game_id=300` correctly, BUT for HAVING/ORDER BY after GROUP BY: body had `c.category ORDER BY ...` and `^` skipped the ORDER BY match (since ORDER is NOT at start).

Fix: removed `^` anchor — search anywhere in body for next clause keyword.

### 5.6 Bug: EXTRACT regression (Bài 2)

Bài 2 SQL: `SELECT username, address_city, address_dist, (EXTRACT(YEAR FROM CURRENT_DATE) - birth_year) ... FROM ...` — naive fromIdx tìm FIRST "FROM" keyword, match "FROM" trong `EXTRACT(YEAR FROM ...)` → hiểu "YEAR" làm table.

Fix: track paren depth; chỉ match FROM tại depth=0.

### 5.7 Bug: alias-prefixed col refs missing trong cross-product

SELECT projection `m.member_name` cần lookup joinedRow key. Cross-product added `members.member_name` (full) + `member_name` (unprefixed) — KHÔNG có `m.member_name` (alias).

Fix: add alias-prefixed keys cross-product (`tableAlias[tableName]` lookup).

### 5.8 Bug: `onConds is not defined` JS ReferenceError

Sau khi restructure parser, tôi re-declared `var onConds = []` thay vì outer var. Trước đó outer scope có `var onConds;`. Trong parser inner scope đã redeclare.

Fix: keep inner `var onConds = []` (matching pattern với `parsedTables`).
